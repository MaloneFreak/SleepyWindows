# SleepyWindows - English (base language)
#
# Format: key = value
# - Lines starting with # are comments.
# - \n is a line break (used in message boxes).
# - {0} and {1} are filled in by the app; keep them. {1:00} means "always two digits" (1h05).
# - To add a language, copy this file to lang/<code>.txt (e.g. sv.txt), translate the values
#   and send a pull request. Missing keys fall back to English.

_name = English
